# Run from the MPEK repository root (EMULaToR/baselines/mpek) with the mldb environment.
PY=/home/da24s023/miniforge3/envs/mldb/bin/python
export CUDA_VISIBLE_DEVICES=0

# Check the assets (Mole-BERT checkpoint, ProtT5)
$PY emulator_bench/prepare_assets.py

# List the 42 runs (7 split folders x 2 tasks x 3 seeds). No work is done.
$PY emulator_bench/launch_parallel_retrain.py --gpus 0 --dry_run

# Build the caches once. ProtT5 vectors identical to CataPro's are copied (read only);
# the other 924 are computed (about 6 min on one GPU). Mole-BERT takes about 1 min.
$PY emulator_bench/cache_embeddings.py --device cuda:0

# All 42 runs, one after the other on GPU 0 (--runs_per_gpu 1 is the fastest layout).
# Estimated time: about 23 h. Run it in a detached screen with a log file.
$PY emulator_bench/launch_parallel_retrain.py --gpus 0 --skip_cache --no_progress

# One task or one split group
$PY emulator_bench/launch_parallel_retrain.py --gpus 0 --skip_cache --tasks kcat --split_groups uniprot_time_splits

# One run
$PY emulator_bench/train_single_split.py --task kcat --split_group random_splits_grouped_sequence --split_name all \
  --train_path <EMULaToR>/data/processed/datasets/kinetic_dataset/kcat/random_splits_grouped_sequence/train.parquet \
  --val_path <...>/val.parquet --test_path <...>/test.parquet \
  --embeddings_dir <EMULaToR>/outputs/baselines/mpek/embeddings \
  --out_dir <EMULaToR>/outputs/baselines/mpek/kcat/random_splits_grouped_sequence/all/seed_666 --seed 666

# Multi-run path: same settings and files, fused kernels (about 2.7x the steps/s of the commands above, see REPORT.md).
# Do not run it on run folders that another launcher is writing. One process per task is possible: --tasks kcat / --tasks km.
$PY emulator_bench/launch_multirun.py --skip_cache --tasks kcat km --group_size 1
# Several runs of a task in one group (slower on this GPU), e.g. 3 seeds of every split folder together:
$PY emulator_bench/launch_multirun.py --skip_cache --tasks kcat --group_size 21

# Summary tables over the seeds
$PY emulator_bench/aggregate_mpek_results.py

# Equivalence tests: CPU only, then also the GPU checks (needs the built caches)
emulator_bench/verify_equivalence.sh
emulator_bench/verify_equivalence.sh --gpu

# Smoke test (tiny): 64 rows, 2 epochs, one task, outputs outside the repo
$PY emulator_bench/launch_parallel_retrain.py --gpus 0 --tasks kcat --split_groups uniprot_time_splits --seeds 666 --epochs 2 --limit_rows 64 --output_dir <scratch_dir>
