# Run from the MPEK repository root (baselines/mpek) after "conda activate mldb".
PY=python

# Check the assets (Mole-BERT checkpoint, ProtT5)
$PY emulator_bench/prepare_assets.py

# Cache all embeddings once (fp32 ProtT5, about 6 GB of GPU memory)
$PY emulator_bench/cache_embeddings.py --device cuda:0

# All tasks (kcat km), all default splits, seeds 666 777 888, GPU 1 with 4 runs at once
$PY emulator_bench/launch_parallel_retrain.py --gpus 1 --runs_per_gpu 4

# One run at a time instead
CUDA_VISIBLE_DEVICES=1 $PY emulator_bench/run_split_benchmarks.py --device cuda:0 --cache_device cuda:0

# One task or one split group
$PY emulator_bench/launch_parallel_retrain.py --gpus 1 --runs_per_gpu 4 --tasks kcat --split_groups uniprot_time_splits

# Other data or output folders
$PY emulator_bench/launch_parallel_retrain.py --gpus 0 1 --base_dir <kinetic_dataset> --output_dir <output_dir>

# Summary tables over the seeds
$PY emulator_bench/aggregate_mpek_results.py

# Smoke test (tiny): 64 rows, 2 epochs, one task
$PY emulator_bench/launch_parallel_retrain.py --gpus 1 --tasks kcat --seeds 666 --epochs 2 --limit_rows 64 --output_dir <scratch_dir>
