# Run from the MPEK repository root (baselines/mpek) after "conda activate mldb".
PY=python

# Once: check the assets and cache the embeddings (shared with emulator_bench)
$PY emulator_bench/prepare_assets.py
$PY emulator_bench/cache_embeddings.py --device cuda:0 --tasks kcat

# Both modes, P-seq and Time splits, seeds 666 777 888, GPU 1 with 4 runs at once
$PY assay_emulator_bench/run_split_benchmarks.py --gpus 1 --runs_per_gpu 4

# One mode, or more split groups
$PY assay_emulator_bench/run_split_benchmarks.py --gpus 1 --modes with_conditions
$PY assay_emulator_bench/run_split_benchmarks.py --gpus 1 --split_groups random_splits_grouped_sequence random_splits_grouped_smiles substrate_splits

# Summary tables over the seeds
$PY assay_emulator_bench/aggregate_mpek_results.py

# Tests
$PY -m pytest assay_emulator_bench/tests

# Smoke test (tiny): 64 rows, 2 epochs, one seed
$PY assay_emulator_bench/run_split_benchmarks.py --gpus 1 --seeds 666 --epochs 2 --limit_rows 64 --output_dir <scratch_dir>
